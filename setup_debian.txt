
# I recommend setting up a Debian Chroot to use pip for
# ortools without destroying the Debian package system


mkdir -p /chroots/trixie
debootstrap trixie /chroots/trixie http://deb.debian.org/debian
cd /chroots/trixie/etc
mkdir old
mv passwd passwd- group group- shadow shadow- old
ln /etc/passwd .
ln /etc/passwd- .
ln /etc/group .
ln /etc/group- .
ln /etc/shadow .
ln /etc/shadow- .

chroot /chroots/trixie

cat > ./usr/sbin/policy-rc.d <<EOF
#!/bin/sh
exit 101
EOF
chmod a+x ./usr/sbin/policy-rc.d

exit

# Change fstab in main OS

nano /etc/fstab
#Add
/dev     /chroots/trixie/dev     auto bind 0 0
/dev/pts /chroots/trixie/dev/pts auto bind 0 0
/dev/shm /chroots/trixie/dev/shm auto bind 0 0
/proc    /chroots/trixie/proc    auto bind 0 0
/home    /chroots/trixie/home    auto bind 0 0

#Save and exit

mount -a
chroot /chroots/trixie

apt update
